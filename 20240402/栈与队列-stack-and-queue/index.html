顺序栈

typedef struct
{
    ElemType data[MAXSIZE];
    int top;
} SqStack;
Status InitStack(SqStack *S)
{
    S->top = -1;
    return OK;
}

Status ClearStack(SqStack *S)
{
    S->top = -1;
    return OK;
}

Status StackEmpty(const SqStack *S)
{
    if (S->top >= 0)
        return FALSE;
    return TRUE;
}

Status GetTop(const SqStack *S, ElemType *E)
{
    if (StackEmpty(S))
        return ERROR;
    *E = S->data[S->top];
    return OK;
}

Status Push(SqStack *S, const ElemType *E)
{
    if (S->top >= MAXSIZE - 1)
        return ERROR;
    S->data[++S->top] = *E;
    return OK;
}

Status Pop(SqStack *S, ElemType *E)
{
    if (StackEmpty(S))
        return ERROR;
    *E = S->data[S->top--];
    return OK;
}

int StackLength(const SqStack *S)
{
    return S->top + 1;
}

链栈

typedef struct SNode
{
    ElemType data;
    struct SNode *link;
} SNode;

typedef struct
{
    SNode *top;
} LinkStack;
Status InitStack(LinkStack *S)
{
    S->top = NULL;
    return OK;
}

Status ClearStack(LinkStack *S)
{
    SNode *p = S->top;
    while (p)
    {
        SNode *tmp = p;
        p = p->link;
        free(tmp);
    }
    S->top = NULL;
    return OK;
}

Status StackEmpty(const LinkStack *S)
{
    if (S->top == NULL)
        return TRUE;
    return FALSE;
}

Status GetTop(const LinkStack *S, ElemType *E)
{
    if (StackEmpty(S))
        return ERROR;
    *E = S->top->data;
    return OK;
}

Status Push(LinkStack *S, const ElemType *E)
{
    SNode *p = (SNode *)malloc(sizeof(SNode));
    p->data = *E;
    p->link = S->top;
    S->top = p;
    return OK;
}

Status Pop(LinkStack *S, ElemType *E)
{
    if (StackEmpty(S))
        return ERROR;
    SNode *p = S->top;
    *E = p->data;
    S->top = p->link;
    free(p);
    return OK;
}

int StackLength(LinkStack *S)
{
    SNode *p = S->top;
    int count = 0;
    while (p)
    {
        p = p->link;
        count++;
    }
    return count;
}

顺序队列

typedef struct
{
    ElemType data[MAXSIZE];
    int front;
    int rear;
} SqQueue;
Status InitQueue(SqQueue *Q)
{
    Q->front = Q->rear = 0;
    return OK;
}

Status ClearQueue(SqQueue *Q)
{
    Q->front = Q->rear = 0;
    return OK;
}

Status QueueEmpty(const SqQueue *Q)
{
    if (Q->front == Q->rear)
        return TRUE;
    return FALSE;
}

int QueueLength(const SqQueue *Q)
{
    return (Q->rear - Q->front + MAXSIZE) % MAXSIZE;
}

Status GetHead(const SqQueue *Q, ElemType *E)
{
    if (QueueEmpty(Q))
        return ERROR;
    *E = Q->data[Q->front];
    return OK;
}

Status EnQueue(SqQueue *Q, const ElemType *E)
{
    if ((Q->rear + MAXSIZE + 1) % MAXSIZE == Q->front)
        return ERROR;
    Q->data[Q->rear] = *E;
    Q->rear = (Q->rear + 1) % MAXSIZE;
    return OK;
}

Status DeQueue(SqQueue *Q, ElemType *E)
{
    if (QueueEmpty(Q))
        return ERROR;
    *E = Q->data[Q->front];
    Q->front = (Q->front + 1) % MAXSIZE;
    return OK;
}

链队列

typedef struct QNode
{
    ElemType data;
    struct QNode *next;
} QNode;

typedef struct
{
    QNode *front;
    QNode *rear;
} LinkQueue;
Status InitQueue(LinkQueue *Q)
{
    Q->front = Q->rear = (QNode *)malloc(sizeof(QNode));
    Q->rear->next = NULL;
    return OK;
}

Status ClearQueue(LinkQueue *Q)
{
    QNode *p = Q->front->next;
    while (p)
    {
        QNode *tmp = p->next;
        p = p->next;
        free(tmp);
    }
    return OK;
}

Status QueueEmpty(const LinkQueue *Q)
{
    if (Q->front->next == NULL)
        return TRUE;
    return FALSE;
}

int QueueLength(const LinkQueue *Q)
{
    QNode *p = Q->front;
    int count = 0;
    while (p != Q->rear)
    {
        p = p->next;
        count++;
    }
    return count;
}

Status GetHead(const LinkQueue *Q, ElemType *E)
{
    if (QueueEmpty(Q))
        return ERROR;
    *E = Q->front->next->data;
    return OK;
}

Status EnQueue(LinkQueue *Q, const ElemType *E)
{
    Q->rear->next = (QNode *)malloc(sizeof(QNode));
    Q->rear = Q->rear->next;
    Q->rear->data = *E;
    return OK;
}

Status DeQueue(LinkQueue *Q, ElemType *E)
{
    if (QueueEmpty(Q))
        return ERROR;
    QNode *p = Q->front->next;
    *E = p->data;
    Q->front->next = p->next;
    free(p);
    return OK;
}
